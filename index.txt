<script>
/* ===== Данные (в памяти; на реальном бэкенде это БД) ===== */
const LIMIT = 5000, DAY = 24*3600*1000, NEW_GIFT_PRICE = 150;
const GIFTS = {
  new:  {name:'Новый подарок', price:NEW_GIFT_PRICE, max:20, icon:'🎁'},
  old1: {name:'Старый подарок 1 (эксклюзив)', price:100, max:1, icon:'🧸'},
  old2: {name:'Старый подарок 2 (эксклюзив)', price:200, max:1, icon:'💎'}
};
const users = [
  {id:1, nick:'Алина',  username:'aa7',    ava:'🦊', role:'owner', balance:0,   status:'active', gifts:[]},
  {id:2, nick:'Борис',  username:'boris',  ava:'🐻', role:'admin', balance:300, status:'active', gifts:[]},
  {id:3, nick:'Вера',   username:'vera_k', ava:'🦉', role:'user',  balance:50,  status:'active', gifts:[{uid:1,gid:'new',level:3}]},
  {id:4, nick:'Глеб',   username:'gleb',   ava:'🐺', role:'user',  balance:120, status:'active', gifts:[]},
  {id:5, nick:'Дина',   username:'din',    ava:'🐙', role:'user',  balance:0,   status:'active', gifts:[{uid:2,gid:'old1',level:1}]}
];
let giftSeq = 10, actorId = 1, tab = 'chats';
const ledger = [];          // {to, amt, by, t}
const snapshots = {};       // userId -> копия данных до удаления
const seen = new Set();
const bgs = ['linear-gradient(160deg,#3b2a8c,#c4509b)','linear-gradient(160deg,#0f6e6e,#2a3a8c)','linear-gradient(160deg,#8c3a2a,#e0a030)'];
const storyData = {1:['Привет из истории','Второй кадр'], 3:['Новый день'], 4:['Гуляем','Закат','Домой'], 5:['Мой кот']};

const $ = s => document.querySelector(s);
const esc = s => String(s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
const byId = id => users.find(u => u.id === id);
const actor = () => byId(actorId);
const isOwner = () => actor().role === 'owner';
const isStaff = () => ['owner','admin'].includes(actor().role);

function toast(msg, err){
  const t = $('#toast'); t.textContent = msg; t.className = 'on' + (err ? ' err' : '');
  clearTimeout(toast.t); toast.t = setTimeout(() => t.className = '', 2800);
}
const unameHtml = u => {
  if (u.status === 'deleted') return '<span class="deleted">—</span>';
  const cls = u.username.length === 3 ? 'u3' : '';
  return `<span class="${cls}">@${esc(u.username)}</span>`;
};
const nickHtml = u => u.status === 'deleted' ? '<span class="deleted">DELETE ACCOUNT</span>' : esc(u.nick);
const avaHtml = (u, size) => `<div class="av" style="width:${size}px;height:${size}px">${u.status === 'deleted' ? '👤' : u.ava}</div>`;

/* ===== Чаты + истории ===== */
function renderChats(){
  const withStories = users.filter(u => storyData[u.id] && u.status === 'active');
  let h = '<div class="stories">';
  withStories.forEach(u => {
    h += `<button class="story" data-s="${u.id}"><div class="ring ${seen.has(u.id)?'seen':''}">${avaHtml(u,58).replace('class="av"','class="av" ')}</div><small>${esc(u.nick)}</small></button>`;
  });
  h += '</div>';
  users.filter(u => u.id !== actorId).forEach(u => {
    h += `<div class="row">${avaHtml(u,48)}<div class="txt"><div class="name">${nickHtml(u)}</div><div class="sub">${unameHtml(u)}</div></div></div><div class="sep"></div>`;
  });
  $('#main').innerHTML = h;
  document.querySelectorAll('.story').forEach(b => b.onclick = () => openViewer(withStories.map(u=>u.id), +b.dataset.s));
}

/* ===== Просмотрщик историй ===== */
const V = {order:[], ui:0, si:0, t0:0, raf:0};
const SLIDE_MS = 5000;
function openViewer(order, id){ V.order = order; V.ui = order.indexOf(id); V.si = 0; $('#viewer').classList.add('on'); showSlide(); }
function closeViewer(){ cancelAnimationFrame(V.raf); $('#viewer').classList.remove('on'); renderChats(); }
function showSlide(){
  const uid = V.order[V.ui], u = byId(uid), slides = storyData[uid];
  seen.add(uid);
  $('#slide').style.background = bgs[(uid + V.si) % bgs.length];
  $('#vtext').textContent = slides[V.si];
  $('#vname').textContent = u.nick; $('#vav').textContent = u.ava;
  $('#bars').innerHTML = slides.map(() => '<b><i></i></b>').join('');
  [...$('#bars').children].forEach((b,i) => { if (i < V.si) b.firstChild.style.width = '100%'; });
  V.t0 = performance.now(); cancelAnimationFrame(V.raf);
  const tick = now => {
    const p = Math.min((now - V.t0) / SLIDE_MS, 1);
    $('#bars').children[V.si].firstChild.style.width = p * 100 + '%';
    if (p >= 1) return next(); V.raf = requestAnimationFrame(tick);
  };
  V.raf = requestAnimationFrame(tick);
}
function next(){ const n = storyData[V.order[V.ui]].length; if (V.si < n-1){V.si++;showSlide();} else nextUser(); }
function prev(){ if (V.si > 0){V.si--;showSlide();} else if (V.ui > 0){V.ui--;V.si=0;showSlide();} else showSlide(); }
function nextUser(){ if (V.ui < V.order.length-1){V.ui++;V.si=0;showSlide();} else closeViewer(); }
function prevUser(){ if (V.ui > 0){V.ui--;V.si=0;showSlide();} }
$('#vclose').onclick = closeViewer;
let px, py, pt;
$('#slide').addEventListener('pointerdown', e => { px = e.clientX; py = e.clientY; pt = performance.now(); });
$('#slide').addEventListener('pointerup', e => {
  if (e.target.closest('button')) return;
  const dx = e.clientX - px, dy = e.clientY - py;
  if (Math.abs(dy) > 80 && dy > 0 && Math.abs(dy) > Math.abs(dx)) return closeViewer();
  if (Math.abs(dx) > 60) return dx < 0 ? nextUser() : prevUser();
  const r = $('#slide').getBoundingClientRect();
  (e.clientX - r.left) < r.width / 3 ? prev() : next();
});
addEventListener('keydown', e => { if (!$('#viewer').classList.contains('on')) return;
  if (e.key==='Escape') closeViewer(); if (e.key==='ArrowRight') next(); if (e.key==='ArrowLeft') prev(); });

/* ===== Бизнес-логика (зеркало бэкенда) ===== */
function usedToday(adminId){
  const from = Date.now() - DAY;
  return ledger.filter(l => l.by === adminId && l.t > from).reduce((s,l) => s + l.amt, 0);
}
function grantStars(targetId, amount){
  const a = actor();
  if (!isStaff()) return fail(403, 'Недостаточно прав');
  if (!Number.isInteger(amount) || amount <= 0) return fail(400, 'Некорректная сумма');
  if (a.role !== 'owner' && targetId === a.id) return fail(403, 'Admins cannot grant stars to themselves');
  if (a.role !== 'owner'){
    const used = usedToday(a.id);
    if (used + amount > LIMIT) return fail(429, `Лимит ${LIMIT}/24ч превышен (использовано ${used})`);
  }
  byId(targetId).balance += amount;
  ledger.push({to:targetId, amt:amount, by:a.id, t:Date.now()});
  toast(`Выдано ${amount} ★`); render();
}
function grantGift(targetId, gid, level){
  if (!isOwner()) return fail(403, 'Подарки выдаёт только Owner');
  const g = GIFTS[gid]; level = Math.min(Math.max(level,1), g.max);
  byId(targetId).gifts.push({uid:++giftSeq, gid, level});
  toast(`Подарок выдан (ур. ${level})`); render();
}
function sellGift(userId, uid){
  const u = byId(userId), i = u.gifts.findIndex(g => g.uid === uid);
  if (i < 0) return fail(404, 'Подарок не найден');
  const payout = GIFTS[u.gifts[i].gid].price;
  u.gifts.splice(i, 1);            // полностью исчезает из профиля
  u.balance += payout;
  toast(`Продано: +${payout} ★`); render();
}
function deleteAccount(id){
  if (!isOwner()) return fail(403, 'Только Owner');
  const u = byId(id); if (u.id === actorId) return fail(400, 'Нельзя удалить себя');
  snapshots[id] = {...u, gifts:[...u.gifts]};
  Object.assign(u, {status:'deleted', nick:'DELETE ACCOUNT', username:null, ava:null});
  toast('Аккаунт удалён'); render();
}
function restoreAccount(id){
  if (!isOwner()) return fail(403, 'Только Owner');
  const s = snapshots[id];
  const taken = users.some(o => o.id !== id && o.status !== 'deleted' && o.username && o.username.toLowerCase() === s.username.toLowerCase());
  if (taken) return fail(409, 'Логин уже занят новым пользователем — восстановление невозможно');
  Object.assign(byId(id), {status:'active', nick:s.nick, username:s.username, ava:s.ava});
  delete snapshots[id]; toast('Аккаунт восстановлен'); render();
}
function registerUsername(name){            // имитация регистрации нового человека на освободившийся логин
  const u = users.find(x => x.status === 'deleted' && snapshots[x.id] && snapshots[x.id].username === name);
  if (!u) return;
  users.push({id:users.length+1, nick:'Новичок', username:name, ava:'🐣', role:'user', balance:0, status:'active', gifts:[]});
  toast(`Новый пользователь занял @${name}`); render();
}
function setUsername(id, name){
  name = name.trim().replace(/^@/, '');
  if (!/^[a-z0-9_]{3,32}$/i.test(name)) return fail(400, 'Допустимы a–z, 0–9, _ (от 3 символов)');
  if (name.length === 3 && !isOwner()) return fail(403, '3-значные юзернеймы выдаёт только Owner');
  if (users.some(o => o.id !== id && o.status !== 'deleted' && o.username && o.username.toLowerCase() === name.toLowerCase()))
    return fail(409, 'Юзернейм занят');
  byId(id).username = name; toast('Юзернейм обновлён'); render();
}
function fail(code, msg){ toast(`${code}: ${msg}`, true); }

/* ===== Админка ===== */
function renderAdmin(){
  const a = actor();
  if (!isStaff()){ $('#main').innerHTML = '<div class="card"><h2>Нет доступа</h2><p class="hint">Переключите роль вверху на Owner или Admin.</p></div>'; return; }
  const opts = users.filter(u => u.status === 'active').map(u => `<option value="${u.id}">${esc(u.nick)}${u.id===a.id?' (вы)':''}</option>`).join('');
  const used = a.role === 'admin' ? usedToday(a.id) : 0;
  let h = `<div class="card"><h2>Выдать звёзды ★</h2><div class="form">
    <select id="gs-target">${opts}</select><input id="gs-amt" type="number" min="1" value="1000" aria-label="Количество">
    <button id="gs-go">Выдать</button></div>
    ${a.role==='admin'
      ? `<div class="meter"><i style="width:${Math.min(used/LIMIT*100,100)}%"></i></div><p class="hint">Лимит за 24 ч: ${used} / ${LIMIT}. Себе выдавать нельзя.</p>`
      : '<p class="hint">Owner: без лимита, можно и себе.</p>'}</div>`;
  if (isOwner()){
    h += `<div class="card"><h2>Выдать подарок</h2><div class="form">
      <select id="gg-target">${opts}</select>
      <select id="gg-gift">${Object.entries(GIFTS).map(([k,g])=>`<option value="${k}">${g.icon} ${esc(g.name)}</option>`).join('')}</select>
      <input id="gg-level" type="number" min="1" max="20" value="1" aria-label="Уровень">
      <button id="gg-go">Выдать</button></div>
      <p class="hint">Новый подарок: базовая цена ${NEW_GIFT_PRICE} ★, уровней 1–20. Купить его нельзя.</p></div>
    <div class="card"><h2>3-значный юзернейм</h2><div class="form">
      <select id="un-target">${opts}</select><input id="un-name" placeholder="например, xyz" maxlength="32">
      <button id="un-go">Установить</button></div></div>`;
  }
  h += '<div class="card"><h2>Пользователи</h2>';
  users.forEach(u => {
    const deleted = u.status === 'deleted';
    h += `<div style="padding:10px 0;border-top:1px solid var(--line)">
      <div style="display:flex;align-items:center;gap:10px">${avaHtml(u,40)}
        <div style="flex:1;min-width:0"><div class="name">${nickHtml(u)} <span class="role">${u.role}</span></div>
        <div class="sub">${unameHtml(u)} · <span class="stars">${u.balance} ★</span></div></div>
        ${isOwner() && u.id !== a.id ? (deleted
          ? `<button class="ok" data-restore="${u.id}">Вернуть</button>`
          : `<button class="danger" data-del="${u.id}">Удалить</button>`) : ''}
      </div>
      <div>${u.gifts.map(g => `<span class="gift">${GIFTS[g.gid].icon} ${esc(GIFTS[g.gid].name.split(' (')[0])}${GIFTS[g.gid].max>1?' · ур.'+g.level:''}
         <button class="ghost" data-sell="${u.id}:${g.uid}">Продать +${GIFTS[g.gid].price}★</button></span>`).join('')}</div>
      ${deleted ? `<button class="ghost" style="margin-top:6px;font-size:12px" data-reg="${esc(snapshots[u.id].username)}">Имитировать: кто-то занял @${esc(snapshots[u.id].username)}</button>` : ''}
    </div>`;
  });
  $('#main').innerHTML = h + '</div>';
  $('#gs-go').onclick = () => grantStars(+$('#gs-target').value, parseInt($('#gs-amt').value, 10));
  if (isOwner()){
    $('#gg-go').onclick = () => grantGift(+$('#gg-target').value, $('#gg-gift').value, parseInt($('#gg-level').value,10)||1);
    $('#un-go').onclick = () => setUsername(+$('#un-target').value, $('#un-name').value);
  }
  document.querySelectorAll('[data-del]').forEach(b => b.onclick = () => deleteAccount(+b.dataset.del));
  document.querySelectorAll('[data-restore]').forEach(b => b.onclick = () => restoreAccount(+b.dataset.restore));
  document.querySelectorAll('[data-sell]').forEach(b => b.onclick = () => { const [i,g] = b.dataset.sell.split(':'); sellGift(+i, +g); });
  document.querySelectorAll('[data-reg]').forEach(b => b.onclick = () => registerUsername(b.dataset.reg));
}

/* ===== Каркас ===== */
function render(){ tab === 'chats' ? renderChats() : renderAdmin(); }
function setTab(t){ tab = t; $('#t-chats').className = t==='chats'?'on':''; $('#t-admin').className = t==='admin'?'on':''; render(); }
$('#t-chats').onclick = () => setTab('chats');
$('#t-admin').onclick = () => setTab('admin');
$('#actor').innerHTML = [[1,'Owner (Алина)'],[2,'Admin (Борис)'],[3,'Пользователь (Вера)']].map(([i,n]) => `<option value="${i}">${n}</option>`).join('');
$('#actor').onchange = e => { actorId = +e.target.value; render(); };
render();
